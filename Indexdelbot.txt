// UTILIDADES
function randInt(min, max) {
  return Math.floor(Math.random() * (max - min + 1)) + min;
}
function normalizeName(value) {
  return String(value || '')
    .toLowerCase()
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .replace(/[^a-z0-9]+/g, '');
}
function findDivine(input) {
  const normalized = normalizeName(input);
  if (!normalized) return null;
  return config.divines.find(divine => {
    const names = [
      divine.key,
      divine.name,
      divine.display,
      ...(divine.aliases || [])
    ];
    return names.some(name => {
      const candidate = normalizeName(name);
      return candidate && (
        candidate === normalized ||
        candidate.includes(normalized) ||
        normalized.includes(candidate)
      );
    });
  }) || null;
}
function getDivineDisplay(divine) {
  return divine.display || divine.name || divine.key;
}
function buildUtcTimestamp(dateTime) {
  return Math.floor(dateTime.toUTC().toSeconds());
}

// PERMISOS DE OWNER
function hasOwnerRole(member) {
  return Boolean(
    member?.roles?.cache?.some(role =>
      OWNER_ROLE_IDS.has(role.id)
    )
  );
}
async function ensureOwner(interaction) {
  if (!interaction.guild) {
    await interaction.reply({
      content: '❌ Este comando solo se puede usar en un servidor.',
      ephemeral: true
    });
    return false;
  }
  let member = interaction.guild.members.cache.get(
    interaction.user.id
  );
  if (!member) {
    member = await interaction.guild.members
      .fetch(interaction.user.id)
      .catch(() => null);
  }
  if (!hasOwnerRole(member)) {
    const payload = {
      content:
        '⛔ No tenés permiso. Solo los roles Madaf4kin Owner y Madaf4kin Ownaa pueden utilizar esta función.',
      ephemeral: true
    };
    if (interaction.replied || interaction.deferred) {
      await interaction.followUp(payload).catch(() => {});
    } else {
      await interaction.reply(payload).catch(() => {});
    }
    return false;
  }
  return true;
}
// LECTURA DE SENZ V2
async function getSenzLastSeen(client) {
  const channel = await client.channels.fetch(SENZ_CHANNEL_ID);
  if (!channel || !channel.isTextBased() || !channel.messages) {
    throw new Error('No puedo acceder al canal de SenZ V2.');
  }
  const messages = await channel.messages.fetch({ limit: 100 });
  function extractText(components) {
    const parts = [];
    for (const component of components || []) {
      if (typeof component.content === 'string') {
        parts.push(component.content);
      }
      if (Array.isArray(component.components)) {
        parts.push(extractText(component.components));
      }
    }
    return parts.join('\n');
  }
  const candidates = [...messages.values()]
    .map(message => {
      const content = [
        message.content || '',
        ...message.embeds.map(embed => [
          embed.title || '',
          embed.description || '',
          ...(embed.fields || []).flatMap(field => [
            field.name || '',
            field.value || ''
          ])
        ].join('\n')),
        extractText(message.components)
      ].join('\n');
      return { message, content };
    })
    .filter(item =>
      /Divine\s*[-—–]\s*Last\s*Seen/i.test(item.content)
    )
    .sort(
      (a, b) =>
        b.message.createdTimestamp - a.message.createdTimestamp
    );
  if (candidates.length === 0) {
    throw new Error(
      'No encontré Divine — Last Seen en los últimos 100 mensajes accesibles de SenZ.'
    );
  }
  for (const candidate of candidates) {
    const content = candidate.content;
    const headerRegex = /###\s*.*?Divine\s*[-—–]\s*Last\s*Seen/i;
    const headerMatch = headerRegex.exec(content);
    if (!headerMatch) continue;
    const afterHeader = content.slice(
      headerMatch.index + headerMatch[0].length
    );
    const nextSection = /\n\s*#{1,6}\s+/g.exec(afterHeader);
    const divineSection = nextSection
      ? afterHeader.slice(0, nextSection.index)
      : afterHeader;
    const result = new Map();
    const lineRegex =
      /^\s*.*?([A-Za-z][A-Za-z ]*?)\s*[—–-]\s*<t:(\d+):[tTdDfFR]>\s*$/gm;
    for (const match of divineSection.matchAll(lineRegex)) {
      const name = match[1].trim();
      const timestamp = Number(match[2]);
      const normalized = normalizeName(name);
      const canonical = SENZ_DIVINE_NAMES.find(
        item => normalizeName(item) === normalized
      );
      if (
        canonical &&
        Number.isSafeInteger(timestamp) &&
        timestamp > 0
      ) {
        result.set(canonical, timestamp);
      }
    }
    const missing = SENZ_DIVINE_NAMES.filter(
      name => !result.has(name)
    );
    if (missing.length === 0) {
      return result;
    }
  }
  throw new Error(
    'No se encontraron todos los Divines en los componentes de SenZ. ' +
    'Faltan: ' + SENZ_DIVINE_NAMES.filter(name => {
      return !candidates.some(candidate => {
        const text = candidate.content;
        return text.includes(name);
      });
    }).join(', ') +
    '. No se publicó la predicción.'
  );
}
// PROBABILIDADES SEGÚN EL TIEMPO DESDE LA ÚLTIMA APARICIÓN
function calculateSenzChance(divineName, lastSeenTimestamp) {
  const elapsedSeconds = Math.max(
    0,
    Math.floor(Date.now() / 1000) - lastSeenTimestamp
  );
  const elapsedDays = elapsedSeconds / 86400;
  const name = normalizeName(divineName);
  const specialDivines = [
    'nightflame',
    'kitsune',
    'unicorn'
  ];
  const isSpecial = specialDivines.includes(name);
  if (elapsedDays < 1) return 11;
  if (elapsedDays < 2) return 20;
  if (elapsedDays < 3) return 25;
  if (elapsedDays < 4) {
    return name === 'kitsune' || name === 'unicorn'
      ? 40
      : 35;
  }
  if (elapsedDays < 5) {
    return isSpecial ? 50 : 48;
  }
  if (!isSpecial) {
    return 48;
  }
  const extraDays = Math.floor(elapsedDays) - 5;
  return Math.min(89, 60 + extraDays * 5);
}
function getSenzDivineObjects() {
  return SENZ_DIVINE_NAMES.map(senzName => {
    const divine = config.divines.find(item => {
      const names = [
        item.key,
        item.name,
        item.display,
        ...(item.aliases || [])
      ].filter(Boolean);
      return names.some(name =>
        normalizeName(name) === normalizeName(senzName)
      );
    });
    if (!divine) {
      throw new Error(
        `No encontré "${senzName}" en config.json. Revisá su nombre o aliases.`
      );
    }
    return divine;
  });
}
function getSenzDivineChance(divine, lastSeenData) {
  const possibleNames = [
    divine.key,
    divine.name,
    divine.display,
    ...(divine.aliases || [])
  ].filter(Boolean);
  const matchingName = SENZ_DIVINE_NAMES.find(senzName =>
    possibleNames.some(name =>
      normalizeName(name) === normalizeName(senzName)
    )
  );
  if (!matchingName || !lastSeenData.has(matchingName)) {
    throw new Error(
      `No hay datos válidos de SenZ para ${getDivineDisplay(divine)}.`
    );
  }
  return calculateSenzChance(
    matchingName,
    lastSeenData.get(matchingName)
  );
}
// HORARIOS DE LAS PREDICCIONES AUTOMÁTICAS
function randomPredictionTimes() {
  const now = DateTime.now().setZone(TIMEZONE);
  let lower = now.plus({
    hours: config.prediction.scheduleMinHoursAhead || 0
  });
  if (lower <= now) {
    lower = now.plus({ minutes: 1 });
  }
  lower = lower
    .set({ second: 0, millisecond: 0 })
    .plus({ minutes: (5 - (lower.minute % 5)) % 5 });
  if (lower <= now) {
    lower = lower.plus({ minutes: 5 });
  }
  const endOfDay = now.endOf('day');
  const available = [];
  for (
    let candidate = lower;
    candidate <= endOfDay;
    candidate = candidate.plus({ minutes: 5 })
  ) {
    if (candidate > now) {
      available.push(candidate);
    }
  }
  if (available.length < 5) {
    throw new Error(
      'No hay suficientes horarios de 5 minutos disponibles para hoy.'
    );
  }
  const chosen = [];
  while (chosen.length < 5) {
    const index = randInt(0, available.length - 1);
    chosen.push(available.splice(index, 1)[0]);
  }
  return chosen.sort((a, b) => a.toMillis() - b.toMillis());
}
function randomFiveDivines() {
  const available = getSenzDivineObjects();
  const selected = [];
  while (selected.length < 5) {
    const index = randInt(0, available.length - 1);
    selected.push(available.splice(index, 1)[0]);
  }
  return selected;
}
// EMBED DE LA PREDICCIÓN AUTOMÁTICA
function getPredictionLines(prediction) {
  return prediction
    .slice()
    .sort((a, b) => a.timestamp - b.timestamp)
    .map(item => {
      const divine = findDivine(item.key);
      if (!divine) {
        return `**${item.key}** · ${item.chance}% · <t:${item.timestamp}:R>`;
      }
      const display = getDivineDisplay(divine);
      const zone = divine.zone
        ? ` · ${divine.zoneEmoji ? divine.zoneEmoji + ' ' : ''}${divine.zone}`
        : '';
      return `${display}${zone} · **${item.chance}%** · <t:${item.timestamp}:R>`;
    });
}
function buildEmbed(prediction) {
  return new EmbedBuilder()
    .setColor(0x7C3AED)
    .setTitle(config.prediction.title || '🥚 SAE PREDICTION')
    .setDescription([
      config.prediction.header || '**Predicciones de hoy:**',
      '',
      ...getPredictionLines(prediction)
    ].join('\n'))
    .setFooter({
      text: 'SAE PREDICTION · Las predicciones pueden fallar.'
    })
    .setTimestamp(new Date());
}

// PUBLICAR PREDICCIÓN AUTOMÁTICA
async function publishDailyPrediction(client, options = {}) {
  const now = DateTime.now().setZone(TIMEZONE);
  const localDate = now.toISODate();
  if (
    !options.force &&
    state.lastPublishedLocalDate === localDate
  ) {
    return;
  }
  const lastSeenData = await getSenzLastSeen(client);
  const times = randomPredictionTimes();
  const selectedDivines = randomFiveDivines();
  const prediction = selectedDivines
    .map((divine, index) => ({
      key: divine.key,
      timestamp: buildUtcTimestamp(times[index]),
      chance: getSenzDivineChance(divine, lastSeenData)
    }))
    .sort((a, b) => a.timestamp - b.timestamp);
  const channel = await client.channels.fetch(CHANNEL_ID);
  if (!channel || !channel.isTextBased()) {
    throw new Error(
      'PREDICTION_CHANNEL_ID no apunta a un canal de texto.'
    );
  }
  const message = await channel.send({
    content: `<@&${PREDICTION_ROLE_ID}>`,
    embeds: [buildEmbed(prediction)],
    allowedMentions: {
      parse: [],
      roles: [PREDICTION_ROLE_ID]
    }
  });
  await addDivineReactions(message);
  state.lastPublishedLocalDate = localDate;
  state.predictionMessageId = message.id;
  state.predictionChannelId = CHANNEL_ID;
  state.currentPrediction = prediction;
  saveState();
  console.log('Predicción diaria publicada:', message.url);
}
// PARSEO PARA EDITAR PREDICCIONES EXISTENTES
function parseTime(content) {
  const cleaned = content.toLowerCase();
  const match = cleaned.match(/\b(\d{1,2}):(\d{2})\b/);
  if (!match) return null;
  const hour = Number(match[1]);
  const minute = Number(match[2]);
  if (hour > 23 || minute > 59) return null;
  const now = DateTime.now().setZone(TIMEZONE);
  let date = now.startOf('day');
  if (/\bpasado\s+manana\b|\bpasado\s+mañana\b/.test(cleaned)) {
    date = date.plus({ days: 2 });
  } else if (/\bmanana\b|\bmañana\b/.test(cleaned)) {
    date = date.plus({ days: 1 });
  } else {
    const dateMatch = cleaned.match(
      /\b(\d{1,2})[/-](\d{1,2})(?:[/-](\d{2,4}))?\b/
    );
    if (dateMatch) {
      let year = dateMatch[3]
        ? Number(dateMatch[3])
        : now.year;
      if (year < 100) year += 2000;
      date = DateTime.fromObject({
        year,
        month: Number(dateMatch[2]),
        day: Number(dateMatch[1])
      }, {
        zone: TIMEZONE
      }).startOf('day');
      if (!date.isValid) return null;
    } else {
      const candidate = date.set({
        hour,
        minute,
        second: 0,
        millisecond: 0
      });
      return candidate > now
        ? candidate
        : candidate.plus({ days: 1 });
    }
  }
  return date.set({
    hour,
    minute,
    second: 0,
    millisecond: 0
  });
}
function parseChance(content) {
  const match = content.match(/(?:^|\s)(\d{1,3})\s*%/i);
  if (!match) return null;
  const chance = Number(match[1]);
  return chance >= 0 && chance <= 89
    ? chance
    : null;
}
// EDITAR LA PREDICCIÓN AUTOMÁTICA ACTUAL
async function editCurrentPrediction(
  client,
  divine,
  newTime,
  newChance
) {
  const now = DateTime.now().setZone(TIMEZONE);
  const diffHours = newTime.diff(now, 'hours').hours;
  if (
    !state.predictionMessageId ||
    diffHours < 0 ||
    diffHours > 72
  ) {
    return false;
  }
  const channel = await client.channels.fetch(
    state.predictionChannelId || CHANNEL_ID
  );
  if (!channel || !channel.isTextBased()) return false;
  const message = await channel.messages.fetch(
    state.predictionMessageId
  );
  const current = (state.currentPrediction || [])
    .map(item => ({ ...item }));
  const target = current.find(
    item => normalizeName(item.key) === normalizeName(divine.key)
  );
  if (!target) return false;
  target.timestamp = buildUtcTimestamp(newTime);
  if (newChance !== null) {
    target.chance = newChance;
  }
  current.sort((a, b) => a.timestamp - b.timestamp);
  state.currentPrediction = current;
  saveState();
  await message.edit({
    embeds: [buildEmbed(current)]
  });
  return true;
}

function buildDivineSelectRows(userId, page = 0) {
  const totalPages = Math.ceil(
    config.divines.length / DIVINE_PAGE_SIZE
  );
  const start = page * DIVINE_PAGE_SIZE;
  const divines = config.divines.slice(
    start,
    start + DIVINE_PAGE_SIZE
  );
  const select = new StringSelectMenuBuilder()
    .setCustomId(`addprediction_divine:${userId}:${page}`)
    .setPlaceholder('¿Qué personaje querés predecir?')
    .setMinValues(1)
    .setMaxValues(1)
    .addOptions(
      divines.map(divine => ({
        label: String(
          divine.name || divine.display || divine.key
        ).slice(0, 100),
        value: String(divine.key).slice(0, 100),
        description: String(divine.zone || 'Divine').slice(0, 100)
      }))
    );
  const rows = [
    new ActionRowBuilder().addComponents(select)
  ];
  if (totalPages > 1) {
    const buttons = [];
    if (page > 0) {
      buttons.push(
        new ButtonBuilder()
          .setCustomId(`addprediction_page:${userId}:${page - 1}`)
          .setLabel('Anterior')
          .setStyle(ButtonStyle.Secondary)
      );
    }
    if (page < totalPages - 1) {
      buttons.push(
        new ButtonBuilder()
          .setCustomId(`addprediction_page:${userId}:${page + 1}`)
          .setLabel('Siguiente')
          .setStyle(ButtonStyle.Secondary)
      );
    }
    rows.push(new ActionRowBuilder().addComponents(buttons));
  }
  return rows;
}
function parseUpcomingClockTime(value) {
  const match = String(value || '')
    .trim()
    .match(/^([01]\d|2[0-3]):([0-5]\d)$/);
  if (!match) return null;
  const hour = Number(match[1]);
  const minute = Number(match[2]);
  const now = DateTime.now().setZone(TIMEZONE);
  let candidate = now.set({
    hour,
    minute,
    second: 0,
    millisecond: 0
  });
  if (candidate <= now) {
    candidate = candidate.plus({ days: 1 });
  }
  return candidate;
}

async function guardianLog(guild, user, reason, total, sanction) {
  try {
    const channel = await guild.channels.fetch(GUARDIAN_DOCUMENTATION_CHANNEL_ID);
    if (!channel?.isTextBased() || !channel.isSendable()) {
      console.error('BOT GUARDIAN: el canal de documentación no está disponible.');
      return;
    }
    const embed = new EmbedBuilder()
      .setColor(0x475569)
      .setTitle('Registro de moderación')
      .setDescription(
        `Usuario: ${user.tag} (${user.id})\n` +
        `Infracción: ${reason}\n` +
        `Total de advertencias: ${total}\n` +
        `Sanción: ${sanction}`
      )
      .setTimestamp(new Date());
    await channel.send({ embeds: [embed], allowedMentions: { parse: [] } });
  } catch (error) {
    console.error('BOT GUARDIAN: no se pudo registrar el suceso:', error.message);
  }
}
async function guardianAddWarning(guild, user, reason, source = 'manual') {
  const warnings = getGuardianUserRecord(guild.id, user.id);
  warnings.push({ reason, source, createdAt: new Date().toISOString() });
  const total = warnings.length;
  let duration = 0;
  let durationLabel = 'Sin timeout automático';
  if (total >= 30) {
    duration = 24 * 60 * 60 * 1000;
    durationLabel = '24 horas';
  } else if (total >= 6) {
    duration = 60 * 60 * 1000;
    durationLabel = '1 hora';
  } else if (total === 5) {
    duration = 10 * 60 * 1000;
    durationLabel = '10 minutos';
  } else if (total === 3) {
    duration = 60 * 1000;
    durationLabel = '1 minuto';
  }
  let sanction = 'Advertencia registrada.';
  if (duration > 0) {
    const member = await guild.members.fetch(user.id).catch(() => null);
    if (member?.moderatable) {
      try {
        await member.timeout(duration, `BOT GUARDIAN: ${reason}`);
        sanction = `Timeout de ${durationLabel}.`;
      } catch (error) {
        sanction = `No se pudo aplicar timeout de ${durationLabel}: ${error.message}`;
      }
    } else {
      sanction = `Advertencia registrada; no se pudo aplicar timeout de ${durationLabel} por permisos o jerarquía de roles.`;
    }
  }
  saveGuardianWarnings();
  await guardianLog(guild, user, reason, total, sanction);
  return { total, sanction };
}

function isValidHttpUrl(value) {
  if (!value) return true;
  try {
    const parsed = new URL(value);
    return parsed.protocol === 'http:' || parsed.protocol === 'https:';
  } catch {
    return false;
  }
}
const EMBED_COLOR_NAMES = {
  morado: 0x7C3AED,
  violeta: 0x8B5CF6,
  purpura: 0x7C3AED,
  rojo: 0xE53935,
  azul: 0x1E88E5,
  'azul oscuro': 0x0D47A1,
  celeste: 0x29B6F6,
  verde: 0x43A047,
  'verde oscuro': 0x1B5E20,
  amarillo: 0xFDD835,
  naranja: 0xFB8C00,
  rosa: 0xEC407A,
  negro: 0x23272A,
  blanco: 0xFFFFFF,
  gris: 0x95A5A6,
  'gris oscuro': 0x424549,
  turquesa: 0x1ABC9C,
  cian: 0x00BCD4,
  cyan: 0x00BCD4,
  magenta: 0xD81B60,
  dorado: 0xD4AF37,
  oro: 0xD4AF37
};
function resolveEmbedColor(value) {
  if (!value || !value.trim()) return EMBED_COLOR_NAMES.morado;
  const raw = value.trim();
  const hex = raw.startsWith('#') ? raw.slice(1) : raw;
  if (/^[0-9a-fA-F]{6}$/.test(hex)) return Number.parseInt(hex, 16);
  const normalized = raw.toLowerCase()
    .normalize('NFD').replace(/[\u0300-\u036f]/g, '');
  return Object.prototype.hasOwnProperty.call(EMBED_COLOR_NAMES, normalized)
    ? EMBED_COLOR_NAMES[normalized]
    : null;
}
function readEmbedFieldPair(interaction, nameId, valueId, label) {
  const name = interaction.fields.getTextInputValue(nameId).trim();
  const value = interaction.fields.getTextInputValue(valueId).trim();
  if (!name && !value) return { field: null, error: null };
  if (!name || !value) {
    return { field: null, error: `En el campo ${label}, completá tanto el nombre como el valor.` };
  }
  if (name.length > 256 || value.length > 1024) {
    return { field: null, error: `El campo ${label} supera el límite: el nombre admite 256 caracteres y el valor 1024.` };
  }
  return { field: { name, value, inline: false }, error: null };
}

client.on(Events.MessageCreate, async message => {
  try {
    if (
      !message.guild ||
      message.channelId !== SENZ_PING_CHANNEL_ID ||
      !message.author.bot ||
      (client.user && message.author.id === client.user.id)
    ) {
      return;
    }
    const searchableText = [
      message.content || '',
      ...message.embeds.flatMap(embed => [
        embed.title || '',
        embed.description || '',
        ...(embed.fields || []).flatMap(field => [
          field.name || '',
          field.value || ''
        ])
      ])
    ].join('\n');
    const detectedRoleIds = new Set();
    for (const role of message.mentions.roles.values()) {
      if (SENZ_PING_ROLE_IDS.has(role.id)) {
        detectedRoleIds.add(role.id);
      }
    }
    for (const roleId of SENZ_PING_ROLE_IDS) {
      if (
        searchableText.includes(`<@&${roleId}>`) ||
        searchableText.includes(roleId)
      ) {
        detectedRoleIds.add(roleId);
      }
    }
    if (detectedRoleIds.size === 0) return;
    await message.channel.send({
      content: [...detectedRoleIds].map(roleId => `<@&${roleId}>`).join(' '),
      allowedMentions: {
        parse: [],
        roles: [...detectedRoleIds],
        users: [],
        repliedUser: false
      }
    });
    console.log(
      `🔔 Pings de SenZ V2 replicados: ${[...detectedRoleIds].join(', ')}`
    );
  } catch (error) {
    console.error('❌ Error replicando pings de SenZ V2:', error);
  }
});

          `🚫 Anti-@everyone: ${message.author.tag}, uso ${total}, sanción ${step.label}, aplicada=${timeoutApplied}`
        );
      } else {
        const warning = ((total - 1) % 3) + 1;
        const dmText = `⚠️ Usaste @everyone: **${warning}/3 advertencias**.
El mensaje fue eliminado. Si llegás a 3 usos, recibirás un aislamiento temporal. Las advertencias y sanciones quedan guardadas aunque el bot se reinicie.`;
        await message.author.send(dmText).catch(error => {
          console.warn(`⚠️ Anti-@everyone: no pude enviar DM a ${message.author.tag}:`, error.message);
        });
        console.log(
          `⚠️ Anti-@everyone: ${message.author.tag}, advertencia ${warning}/3 (uso ${total}).`
        );
      }
      return;
    }
  }
  const content = message.content.trim();
// PERSONALIDAD DEL SAE PREDICTOR
  const asksOpinion = /\bque\s+opinas(?:\s+de\s+esto)?\s*\\??/i.test(
    content.normalize('NFD').replace(/[\u0300-\u036f]/g, '')
  );
  if (
    client.user &&
    message.mentions.users.has(client.user.id) &&
    asksOpinion
  ) {
    if (hasOwnerRole(message.member)) {
      const respuestas = [
        'Puro cuento 🥱',
        'Que habla ese 😂'
      ];
      const respuesta =
        respuestas[Math.floor(Math.random() * respuestas.length)];
      await message.reply({
        content: respuesta,
        allowedMentions: { repliedUser: false }
      });
    } else {
      await message.reply({
        content: 'No tienes suficiente aura para que te haga caso 💀🥀',
        allowedMentions: { repliedUser: false }
      });
    }
    return;
  }
// Desde aquí, los otros comandos solo son para Owners.
  if (!hasOwnerRole(message.member)) {
    return;
  }
// BORRAR CANALES QUE TERMINEN EN "BASE"
  if (content.toLowerCase() === '!borrarbases') {
    const targets = message.guild.channels.cache.filter(channel =>
      channel.name.toLowerCase().endsWith('base')
    );
    if (targets.size === 0) {
      await message.reply('✅ No hay canales que terminen en "base".');
      return;
    }
    await message.reply(
      `⚠️ Encontré ${targets.size} canal(es) para eliminar.`
    );
    let deleted = 0;
    let failed = 0;
    for (const channel of targets.values()) {
      try {
        await channel.delete(
          'Comando !borrarbases ejecutado por un Owner autorizado'
        );
        deleted++;
      } catch (error) {
        failed++;
        console.error(
          `Error eliminando ${channel.name} (${channel.id}):`,
          error
        );
      }
    }
    await message.channel.send(
      `🧹 **Limpieza terminada**\n` +
      `✅ Eliminados: ${deleted}\n` +
      `❌ Fallidos: ${failed}`
    ).catch(() => {});
    return;
  }
// GENERAR UNA NUEVA PREDICCIÓN AUTOMÁTICA
  if (content.toLowerCase() === '!nuevaprediccion') {
    try {
      await publishDailyPrediction(client, { force: true });
      await message.reply({
        content: 'Nueva predicción subida maestro. <:Colossus:1557197762645467166>',
        allowedMentions: { parse: [] }
      });
    } catch (error) {
      console.error('Error en !nuevaprediccion:', error);
      await message.reply(
        '❌ No pude subir la predicción. Revisá la consola.'
      );
    }
    return;
  }
// EDITAR UN DIVINE DE LA PREDICCIÓN AUTOMÁTICA
  const time = parseTime(content);
  if (!time) return;
  const chance = parseChance(content);
  const divine = findDivine(content);
  if (!divine) return;
  try {
    const changed = await editCurrentPrediction(
      client,
      divine,
      time,
      chance
    );
    if (changed) {
      await message.reply(
        `✅ Predicción de **${getDivineDisplay(divine)}** actualizada.`
      );
    } else {
      await message.reply(
        '⚠️ No encontré ese Divine en la predicción actual o la fecha supera las 72 horas.'
      );
    }
  } catch (error) {
    console.error('Error editando predicción:', error);
    await message.reply('❌ No pude editar la predicción.');
  }
});
